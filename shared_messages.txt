## ⏰ 35–70 Minutes: Location + Firebase Logic

### `app.js`

```javascript
import { initializeApp } from "https://www.gstatic.com/firebasejs/10.7.0/firebase-app.js";
import { getFirestore, addDoc, collection } from "https://www.gstatic.com/firebasejs/10.7.0/firebase-firestore.js";

const firebaseConfig = {
  apiKey: "YOUR_KEY",
  authDomain: "YOUR_PROJECT.firebaseapp.com",
  projectId: "YOUR_PROJECT_ID"
};

const app = initializeApp(firebaseConfig);
const db = getFirestore(app);

function getLocation(callback) {
  navigator.geolocation.getCurrentPosition(
    pos => callback(pos.coords.latitude, pos.coords.longitude),
    err => alert("Location access required")
  );
}

window.sendRequest = function () {
  getLocation((lat, lng) => {
    addDoc(collection(db, "requests"), {
      name: document.getElementById("name").value,
      phone: document.getElementById("phone").value,
      need: document.getElementById("need").value,
      latitude: lat,
      longitude: lng,
      priority: "NORMAL",
      time: Date.now()
    });
    alert("Request sent");
  });
};

window.sendSOS = function () {
  getLocation((lat, lng) => {
    addDoc(collection(db, "requests"), {
      name: null,
      phone: null,
      need: "SOS",
      latitude: lat,
      longitude: lng,
      priority: "HIGH",
      time: Date.now()
    });
    alert("SOS sent");
  });
};
```

# 3️⃣ `app.js` (Firebase + Live Location Logic)

```javascript
import { initializeApp } from "https://www.gstatic.com/firebasejs/10.7.0/firebase-app.js";
import { getFirestore, addDoc, collection } from "https://www.gstatic.com/firebasejs/10.7.0/firebase-firestore.js";

/* 🔴 REPLACE WITH YOUR FIREBASE CONFIG */
const firebaseConfig = {
  apiKey: "YOUR_API_KEY",
  authDomain: "YOUR_PROJECT.firebaseapp.com",
  projectId: "YOUR_PROJECT_ID",
};

const app = initializeApp(firebaseConfig);
const db = getFirestore(app);

/* Get Current Location */
function getLocation(callback) {
  navigator.geolocation.getCurrentPosition(
    pos => callback(pos.coords.latitude, pos.coords.longitude),
    () => alert("Location access is required to request help")
  );
}

/* Normal Request */
window.sendRequest = function () {
  const name = document.getElementById("name").value;
  const phone = document.getElementById("phone").value;
  const need = document.getElementById("need").value;

  if (!need || need.includes("Select")) {
    alert("Please select your requirement");
    return;
  }

  getLocation((lat, lng) => {
    addDoc(collection(db, "requests"), {
      name,
      phone,
      need,
      latitude: lat,
      longitude: lng,
      priority: "NORMAL",
      timestamp: Date.now()
    });
    alert("Help request sent successfully");
  });
};

/* SOS Request */
window.sendSOS = function () {
  getLocation((lat, lng) => {
    addDoc(collection(db, "requests"), {
      name: null,
      phone: null,
      need: "SOS",
      latitude: lat,
      longitude: lng,
      priority: "HIGH",
      timestamp: Date.now()
    });
    alert("SOS alert sent. Help is on the way.");
  });
};
```

import { initializeApp } from "https://www.gstatic.com/firebasejs/10.7.0/firebase-app.js";
import { getFirestore, addDoc, collection } from "https://www.gstatic.com/firebasejs/10.7.0/firebase-firestore.js";

const firebaseConfig = {

const db = getFirestore(app);

function getLocation(statusElement, callback) {
  statusElement.innerText = "📍 Fetching current location...";

  navigator.geolocation.getCurrentPosition(
    pos => {
      statusElement.innerText = "✅ Location detected successfully";
      callback(pos.coords.latitude, pos.coords.longitude);
    },
    () => {
      statusElement.innerText = "❌ Location access denied";
      alert("Location permission is required");
    }
  );
}

window.sendRequest = function () {
  const name = document.getElementById("name").value;
  const phone = document.getElementById("phone").value;

  if (!need || need.includes("Select")) {
    alert("Please select your requirement");
    return;
  }

  getLocation(status, (lat, lng) => {
    addDoc(collection(db, "requests"), {
      name,
      phone,
      need,
      latitude: lat,
      longitude: lng,
      priority: "NORMAL",
      timestamp: Date.now()
    });

};

window.sendSOS = function () {
